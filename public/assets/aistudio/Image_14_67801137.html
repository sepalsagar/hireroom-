type RingBuffer[T any] struct {
    buffer []T
    mask   uint64
    _pad0  [7]uint64 // Cache-line anti false-sharing padding
    head   atomic.Uint64
    _pad1  [7]uint64
    tail   atomic.Uint64
}

// Push attempts non-blocking write with single atomic compare-and-swap
func (r *RingBuffer[T]) Push(item T) bool {
    tail := r.tail.Load()
    head := r.head.Load()
    if (tail - head) >= r.mask+1 {
        return false // Buffer saturated
    }
    r.buffer[tail&r.mask] = item
    r.tail.Store(tail + 1)
    return true
}
